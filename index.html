/* ===================================== QR CODE V4 - CONTROLE DE CACHE ===================================== */ function getCurrentSiteURL() { /* Usa a URL base do site. */ const url = new URL(window.location.href); /* Remove parâmetros antigos. */ url.searchParams.delete("v"); url.searchParams.delete("build"); url.searchParams.delete("cache"); url.searchParams.delete("updated"); /* Identificação da versão. */ url.searchParams.set( "v", SITE_VERSION ); /* BUILD FIXO DA PUBLICAÇÃO. Sempre que você atualizar o arquivo no servidor, altere SITE_BUILD. */ url.searchParams.set( "build", SITE_BUILD ); /* Timestamp adicional. */ url.searchParams.set( "cache", Date.now() ); return url.toString(); } function openQRModal() { document .getElementById("qrModal") .classList.add("active"); generateQRCode(); } function closeQRModal() { document .getElementById("qrModal") .classList.remove("active"); } function generateQRCode() { const container = document.getElementById("qrcode"); const currentURL = getCurrentSiteURL(); /* Mostra a URL exata usada. */ document .getElementById("qrUrlInfo") .textContent = "Versão atual do QR Code: " + SITE_VERSION + " | Build: " + SITE_BUILD; /* Limpa QR Code anterior. */ container.innerHTML = ""; /* Gera novo QR Code. */ new QRCode( container, { text: currentURL, width: 220, height: 220, colorDark: "#000000", colorLight: "#ffffff", correctLevel: QRCode.CorrectLevel.H } ); }
